# Windows Kiro Setup Script with Gandalf MCP # This script automates the installation process for Kiro and Gandalf MCP # Last Updated: 12/17/2025 <# .SYNOPSIS Automated setup script for Kiro and Gandalf MCP on Windows .DESCRIPTION This script automates the installation and configuration of Kiro (Amazon's AI IDE), and Gandalf MCP Server for enhanced Excel/Word capabilities. .NOTES Prerequisites: - US Exports form completed (4 hour wait for propagation) - Required permissions granted by manager (4 hour wait for propagation) - Amazon internal network access with valid Midway authentication #> # Requires PowerShell 5.1 or higher #Requires -Version 5.1 # Set error action preference $ErrorActionPreference = "Stop" # Script configuration $script:LogFile = "$env:USERPROFILE\Desktop\Kiro-Setup-Log.txt" # Color-coded output functions function Write-Header { param([string]$Message) Write-Host "`n========================================" -ForegroundColor Cyan Write-Host $Message -ForegroundColor Cyan Write-Host "========================================`n" -ForegroundColor Cyan Add-Content -Path $script:LogFile -Value "`n[$(Get-Date)] $Message" } function Write-Success { param([string]$Message) Write-Host "[SUCCESS] $Message" -ForegroundColor Green Add-Content -Path $script:LogFile -Value "[SUCCESS] $Message" } function Write-Warning-Custom { param([string]$Message) Write-Host "[WARNING] $Message" -ForegroundColor Yellow Add-Content -Path $script:LogFile -Value "[WARNING] $Message" } function Write-Error-Custom { param([string]$Message) Write-Host "[ERROR] $Message" -ForegroundColor Red Add-Content -Path $script:LogFile -Value "[ERROR] $Message" } function Write-Info { param([string]$Message) Write-Host "[INFO] $Message" -ForegroundColor White Add-Content -Path $script:LogFile -Value "[INFO] $Message" } function Write-Action { param([string]$Message) Write-Host "[ACTION] $Message" -ForegroundColor Magenta Add-Content -Path $script:LogFile -Value "[ACTION] $Message" } # Initialize log file function Initialize-Log { $logHeader = @" ======================================== Kiro Setup Log Started: $(Get-Date) ======================================== "@ Set-Content -Path $script:LogFile -Value $logHeader Write-Info "Log file created at: $script:LogFile" } # Display menu and get user choice function Show-Menu { Clear-Host Write-Host @" ======================================================================== Windows Kiro Setup with Gandalf MCP Installation Menu ======================================================================== "@ -ForegroundColor Cyan Write-Host "Please select what you would like to do:" -ForegroundColor Yellow Write-Host "" Write-Host " 1. Full Installation (Recommended for new users)" Write-Host " - Install Kiro and Gandalf MCP" Write-Host "" Write-Host " 2. Install Gandalf Only" Write-Host " - For users who already have Kiro installed" Write-Host "" Write-Host " 3. Install Excel Theme Only" Write-Host " - Add Excel-inspired theme to existing Kiro installation" Write-Host "" Write-Host " 4. Run Daily Midway Setup" Write-Host " - Execute mwinit authentication and view troubleshooting tips" Write-Host "" Write-Host " 5. Exit" Write-Host "" $choice = Read-Host "Enter your choice (1-5)" return $choice } # Configure Midway authentication function Configure-Midway { Write-Header "SECTION 1: Midway Authentication" # Check OpenSSH installation Write-Info "Verifying OpenSSH installation..." $sshCommand = Get-Command ssh -ErrorAction SilentlyContinue if ($sshCommand) { # Get version info (ssh -V outputs to stderr, so capture it) $sshVersion = (cmd /c "ssh -V 2>&1") Write-Success "OpenSSH is installed: $sshVersion" } else { Write-Error-Custom "OpenSSH is not installed or not in PATH" Write-Info "Please ensure OpenSSH is installed on your system" throw "OpenSSH not found" } # Check if SSH key already exists $sshKeyPath = "$env:USERPROFILE\.ssh\id_ecdsa" $sshKeyExists = Test-Path $sshKeyPath if ($sshKeyExists) { Write-Success "SSH key already exists at: $sshKeyPath" $response = Read-Host "Do you want to regenerate your SSH key? (Y/N)" if ($response -ne 'Y' -and $response -ne 'y') { Write-Info "Using existing SSH key" } else { $sshKeyExists = $false } } if (-not $sshKeyExists) { Write-Action "Generating SSH keypair for Midway authentication..." Write-Info "This SSH key is required for mwinit to create your SSH certificate" try { # Create .ssh directory if it doesn't exist $sshDir = "$env:USERPROFILE\.ssh" if (-not (Test-Path $sshDir)) { New-Item -ItemType Directory -Path $sshDir -Force | Out-Null Write-Info "Created .ssh directory: $sshDir" } # Generate SSH key without passphrase (automated) Write-Info "Generating ECDSA SSH keypair (no passphrase for compatibility)..." $process = Start-Process -FilePath "ssh-keygen" -ArgumentList @( "-t", "ecdsa", "-f", $sshKeyPath, "-N", '""' ) -NoNewWindow -Wait -PassThru if ($process.ExitCode -eq 0) { Write-Success "SSH keypair generated successfully!" Write-Info "Private key: $sshKeyPath" Write-Info "Public key: $sshKeyPath.pub" } else { Write-Error-Custom "SSH key generation failed with exit code: $($process.ExitCode)" throw "SSH key generation failed" } } catch { Write-Error-Custom "Failed to generate SSH keypair: $_" Write-Info "You may need to generate it manually using: ssh-keygen -t ecdsa" throw } } # Run mwinit to authenticate Write-Action "Running Midway authentication (mwinit)..." Write-Warning-Custom "You will be prompted to enter your password" Write-Warning-Custom "Note: When entering your password, you won't see it on screen - this is normal for security" Write-Host "" try { # Check if mwinit is available $mwinitCmd = Get-Command mwinit -ErrorAction SilentlyContinue if (-not $mwinitCmd) { Write-Error-Custom "mwinit command not found" Write-Info "Please ensure you have completed the prerequisites and mwinit is installed" throw "mwinit not available" } Write-Info "Running: mwinit -o -s" Write-Host "Please follow the authentication prompts below:" -ForegroundColor Yellow Write-Host "" # Run mwinit interactively with -o (non-interactive) and -s (short-lived cert) flags & mwinit -o -s if ($LASTEXITCODE -eq 0) { Write-Success "Midway authentication completed successfully!" Write-Info "Your Midway session is valid for 24 hours" } else { Write-Warning-Custom "mwinit exited with code: $LASTEXITCODE" $response = Read-Host "`nDid mwinit complete successfully? (Y/N)" if ($response -ne 'Y' -and $response -ne 'y') { Write-Error-Custom "Midway authentication failed" throw "mwinit authentication failed" } } } catch { Write-Error-Custom "Midway authentication error: $_" Write-Info "You may need to run 'mwinit' manually and then restart this script" throw } Write-Success "Midway authentication completed!" } # Install Kiro function Install-Kiro { Write-Header "SECTION 2: Installing Kiro" Write-Info "Kiro is Amazon's AI-powered IDE" # Step 1: Check and install Toolbox Write-Action "Checking for Toolbox installation..." $toolboxCmd = Get-Command toolbox -ErrorAction SilentlyContinue if (-not $toolboxCmd) { Write-Action "Toolbox not found, installing..." try { $bootstrapPath = "toolbox-bootstrap.cmd" $nonce = Get-Random # Download bootstrap using embedded authentication curl.exe --ssl-no-revoke -X POST ` --data '{\"os\":\"windows\"}' ` -H "Content-Type: application/json" ` -H "Authorization: $(curl.exe --ssl-no-revoke -L ` --cookie $Env:USERPROFILE\.midway\cookie ` --cookie-jar $Env:USERPROFILE\.midway\cookie ` $('https://midway-auth.amazon.com/SSO?client_id=https://us-east-1.prod.release-service.toolbox.builder-tools.aws.dev&response_type=id_token&nonce='+$nonce+'&redirect_uri=https://us-east-1.prod.release-service.toolbox.builder-tools.aws.dev:443'))" ` https://us-east-1.prod.release-service.toolbox.builder-tools.aws.dev/v1/bootstrap ` -o $bootstrapPath if (Test-Path $bootstrapPath) { Write-Success "Toolbox bootstrap downloaded" # Run bootstrap Write-Action "Running Toolbox installation..." & .\$bootstrapPath # Add to PATH Write-Action "Adding Toolbox to PATH..." $path = [Environment]::GetEnvironmentVariable('PATH', 'User') $toolboxPath = "$env:LOCALAPPDATA\Toolbox\bin" if ($path -notlike "*$toolboxPath*") { [Environment]::SetEnvironmentVariable('PATH', $path.TrimEnd(";") + ";$toolboxPath", 'User') $env:PATH = $env:PATH + ";$toolboxPath" } # Clean up bootstrap Remove-Item $bootstrapPath -Force -ErrorAction SilentlyContinue Write-Success "Toolbox installed successfully" } else { throw "Bootstrap download failed" } } catch { Write-Error-Custom "Toolbox installation failed: $_" Write-Info "You may need to install Toolbox manually" throw } } else { Write-Success "Toolbox is already installed" } # Step 2: Install axe Write-Action "Installing axe via Toolbox..." try { & toolbox install axe --force Write-Success "Axe installed successfully!" } catch { Write-Warning-Custom "Axe may already be installed or installation failed: $_" Write-Info "Continuing with Kiro initialization..." } # Step 3: Initialize Kiro Write-Action "Initializing Kiro..." try { $axeCmd = Get-Command axe -ErrorAction SilentlyContinue if (-not $axeCmd) { # Refresh PATH $env:PATH = [Environment]::GetEnvironmentVariable('PATH', 'User') + ";" + [Environment]::GetEnvironmentVariable('PATH', 'Machine') $axeCmd = Get-Command axe -ErrorAction SilentlyContinue } if ($axeCmd) { Write-Info "Running: axe init kiro" & axe init kiro Write-Success "Kiro initialized successfully!" Write-Info "Kiro is now ready to use" } else { Write-Error-Custom "axe command not found after installation" Write-Info "You may need to restart PowerShell and run 'axe init kiro' manually" } } catch { Write-Warning-Custom "Kiro initialization encountered an issue: $_" Write-Info "If axe is installed, try running 'axe init kiro' manually" } Write-Success "Kiro installation completed!" } # Install Gandalf MCP Server function Install-GandalfMCP { Write-Header "SECTION 3: Installing Gandalf MCP Server" Write-Info "Gandalf MCP allows Kiro to interact with Excel and Word files" # Check and install Toolbox if needed $toolboxCmd = Get-Command toolbox -ErrorAction SilentlyContinue if (-not $toolboxCmd) { Write-Error-Custom "Toolbox is required but not found" Write-Info "Please run Full Installation (Option 1) first" throw "Toolbox not available" } # Check if gandalf-mcp-server is already installed Write-Action "Checking for existing Gandalf installation..." $gandalfCmd = Get-Command gandalf-mcp-server -ErrorAction SilentlyContinue if ($gandalfCmd -and $gandalfCmd.Source -match "Toolbox") { Write-Success "Gandalf MCP Server is already installed from Toolbox" } else { # Register DSDI Teams Toolbox Repository Write-Action "Registering DSDI Teams Toolbox repository..." try { & toolbox registry add s3://buildertoolbox-registry-dsditool-registry-us-west-2/tools.json 2>&1 | Out-Null Write-Success "DSDI registry added" } catch { Write-Info "Registry may already be added" } # Install Gandalf MCP Server Write-Action "Installing Gandalf MCP Server via Toolbox..." try { & toolbox install gandalf-mcp-server --force Write-Success "Gandalf MCP Server installed successfully!" # Install Gandalf Aide MCP Server Write-Action "Installing Gandalf Aide MCP Server via Toolbox..." & toolbox install gandalf-aide-mcp-server --force Write-Success "Gandalf Aide MCP Server installed successfully!" } catch { Write-Error-Custom "Gandalf installation failed: $_" throw } } # Configure Kiro MCP Settings Write-Action "Configuring Gandalf in Kiro MCP settings..." $mcpSettingsPath = "$env:USERPROFILE\.kiro\settings\mcp.json" try { # Create directory if it doesn't exist $mcpSettingsDir = Split-Path $mcpSettingsPath -Parent if (-not (Test-Path $mcpSettingsDir)) { New-Item -ItemType Directory -Path $mcpSettingsDir -Force | Out-Null Write-Info "Created Kiro settings directory: $mcpSettingsDir" } # Initialize settings object $mcpSettings = $null # Read existing settings file if it exists if (Test-Path $mcpSettingsPath) { try { $content = Get-Content $mcpSettingsPath -Raw if ($content -and $content.Trim()) { $mcpSettings = $content | ConvertFrom-Json Write-Info "Existing MCP settings found" } } catch { Write-Warning-Custom "Could not parse existing settings file, creating new one" $mcpSettings = $null } } # Create new settings structure if needed if (-not $mcpSettings) { $mcpSettings = New-Object PSObject $mcpSettings | Add-Member -MemberType NoteProperty -Name "mcpServers" -Value (New-Object PSObject) Write-Info "Creating new MCP settings file" } # Ensure mcpServers property exists if (-not ($mcpSettings.PSObject.Properties.Name -contains "mcpServers")) { $mcpSettings | Add-Member -MemberType NoteProperty -Name "mcpServers" -Value (New-Object PSObject) -Force } # Check if Gandalf is already configured if ($mcpSettings.mcpServers.PSObject.Properties.Name -contains "gandalf") { Write-Success "Gandalf MCP Server is already configured in Kiro" } else { # Add Gandalf configuration for Kiro $gandalfConfig = New-Object PSObject $gandalfConfig | Add-Member -MemberType NoteProperty -Name "disabled" -Value $false $gandalfConfig | Add-Member -MemberType NoteProperty -Name "timeout" -Value 300 $gandalfConfig | Add-Member -MemberType NoteProperty -Name "type" -Value "stdio" $gandalfConfig | Add-Member -MemberType NoteProperty -Name "command" -Value "gandalf-mcp-server" $gandalfConfig | Add-Member -MemberType NoteProperty -Name "args" -Value @() $envConfig = New-Object PSObject $envConfig | Add-Member -MemberType NoteProperty -Name "PORT" -Value "8001" $envConfig | Add-Member -MemberType NoteProperty -Name "OFFICE_MCP_SESSION_TTL" -Value "3600" $envConfig | Add-Member -MemberType NoteProperty -Name "OFFICE_MCP_MAX_OPEN" -Value "8" $envConfig | Add-Member -MemberType NoteProperty -Name "OFFICE_MCP_MAX_EXPIRED_HISTORY" -Value "100" $gandalfConfig | Add-Member -MemberType NoteProperty -Name "env" -Value $envConfig $autoApproveList = @("open_excel_workbook", "list_excel_workbooks", "get_excel_workbook_metadata", "read_excel_data") $gandalfConfig | Add-Member -MemberType NoteProperty -Name "autoApprove" -Value $autoApproveList $mcpSettings.mcpServers | Add-Member -MemberType NoteProperty -Name "gandalf" -Value $gandalfConfig -Force # Add Gandalf Aide configuration $gandalfAideConfig = New-Object PSObject $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "disabled" -Value $false $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "timeout" -Value 600 $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "type" -Value "stdio" $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "command" -Value "gandalf-aide-mcp-server" $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "args" -Value @() $gandalfAideConfig | Add-Member -MemberType NoteProperty -Name "autoApprove" -Value @("list_tools") $mcpSettings.mcpServers | Add-Member -MemberType NoteProperty -Name "gandalf-aide" -Value $gandalfAideConfig -Force # Save configuration $jsonSettings = $mcpSettings | ConvertTo-Json -Depth 10 Set-Content -Path $mcpSettingsPath -Value $jsonSettings -Force Write-Success "Gandalf and Gandalf Aide MCP Servers configured in Kiro" Write-Info "Configuration saved to: $mcpSettingsPath" } } catch { Write-Error-Custom "Failed to configure Gandalf in Kiro: $_" Write-Info "You may need to configure manually" throw } Write-Success "Gandalf MCP Server installation and configuration completed!" } # Install Excel Theme (Optional) function Install-ExcelTheme { Write-Header "Excel Theme (Optional)" $response = Read-Host "Do you want to enable the Excel theme for Kiro? (Y/N)" if ($response -ne 'Y' -and $response -ne 'y') { Write-Info "Skipping Excel theme installation" return } Write-Action "Installing Office Theme extension for Kiro..." try { # Try to find kiro command $kiroCmd = "kiro" $kiroFound = Get-Command kiro -ErrorAction SilentlyContinue # If not found, try the direct path if (-not $kiroFound) { $kiroCmdPath = "$env:LOCALAPPDATA\Programs\Kiro\bin\kiro.cmd" if (Test-Path $kiroCmdPath) { $kiroCmd = $kiroCmdPath Write-Info "Using Kiro from: $kiroCmdPath" } else { Write-Warning-Custom "Kiro command not found" Write-Info "You can manually install Office Theme from Kiro's Extensions marketplace" return } } # Install the extension & $kiroCmd --install-extension huacat.office-theme --force 2>&1 | Out-Null Write-Success "Office Theme extension installed successfully!" Write-Info "The Excel-inspired theme is now available in Kiro" Write-Info "To activate: Go to File > Preferences > Color Theme and select an Office theme" } catch { Write-Warning-Custom "Could not install Office Theme extension: $_" Write-Info "You can manually install it from Kiro's Extensions marketplace" Write-Info "Search for 'Office Theme' by huacat" } } # Daily setup reminder function Show-DailySetupReminder { Write-Header "Daily Setup Reminder" Write-Info "Remember: You need to run 'mwinit' EVERY DAY before using Kiro/Gandalf" Write-Host "`nDaily Setup Steps:" Write-Host " 1. Open a terminal (Windows Start -> search 'Terminal')" Write-Host " 2. Run: mwinit -o -s" Write-Host " 3. Follow the authentication prompts" Write-Host " 4. Midway session lasts 24 hours" Write-Warning-Custom "Midway authentication expires after 24 hours!" } # Common issues and troubleshooting function Show-TroubleshootingInfo { Write-Header "Common Issues & Troubleshooting" Write-Info "Throttling/429 Errors:" Write-Host " - You may hit 'traffic jams' when using popular models" Write-Host " - Solution 1: Switch to a different AWS region" Write-Host " - Solution 2: Wait a few minutes and try again" Write-Host " - Solution 3: Use a different model" Write-Info "`nKiro/Axe Issues:" Write-Host " - If 'axe' command not found, restart PowerShell" Write-Host " - Run 'toolbox install axe --force' if needed" Write-Host " - Run 'axe init kiro' manually if initialization failed" Write-Info "`nMidway Issues:" Write-Host " - Run 'mwinit -o -s' if you see authentication errors" Write-Host " - Midway must be refreshed daily" Write-Info "`nGet Help:" Write-Host " - Join Slack channel #gandalf-mcp-interest for support" Write-Host " - Share error logs from: $script:LogFile" } # Main execution function Main { try { # Set PowerShell window to always be on top Add-Type @" using System; using System.Runtime.InteropServices; public class WindowHelper { [DllImport("user32.dll")] public static extern bool SetWindowPos(IntPtr hWnd, IntPtr hWndInsertAfter, int X, int Y, int cx, int cy, uint uFlags); public static readonly IntPtr HWND_TOPMOST = new IntPtr(-1); public const uint SWP_NOMOVE = 0x0002; public const uint SWP_NOSIZE = 0x0001; public const uint SWP_SHOWWINDOW = 0x0040; } "@ try { $consolePtr = [System.Diagnostics.Process]::GetCurrentProcess().MainWindowHandle [WindowHelper]::SetWindowPos($consolePtr, [WindowHelper]::HWND_TOPMOST, 0, 0, 0, 0, [WindowHelper]::SWP_NOMOVE -bor [WindowHelper]::SWP_NOSIZE -bor [WindowHelper]::SWP_SHOWWINDOW) | Out-Null } catch { # If setting window on top fails, continue anyway } # Show menu and get user choice $userChoice = Show-Menu # Initialize log Clear-Host Initialize-Log switch ($userChoice) { "1" { # Full Installation Write-Host "Starting Full Kiro Installation..." -ForegroundColor Cyan Write-Host "" Configure-Midway Install-Kiro Install-GandalfMCP Install-ExcelTheme # Pause before showing daily reminders Write-Host "" Read-Host "Press Enter to view daily setup reminders and tips" Show-DailySetupReminder Show-TroubleshootingInfo # Final summary Write-Header "Setup Complete!" Write-Success "All installation steps completed successfully!" Write-Info "Log file saved to: $script:LogFile" Write-Host "`n*** Congratulations! You're ready to use Kiro with Gandalf MCP! ***" -ForegroundColor Green Write-Host "*** To launch Kiro: ***" -ForegroundColor Cyan Write-Host " - Search 'Kiro' in Windows Start Menu, or" -ForegroundColor Cyan Write-Host " - Open a terminal and run: kiro" -ForegroundColor Cyan } "2" { # Gandalf Only Write-Host "Installing Gandalf MCP Only..." -ForegroundColor Cyan Write-Host "" # Check for mwinit first $mwinitCmd = Get-Command mwinit -ErrorAction SilentlyContinue if (-not $mwinitCmd) { Write-Error-Custom "mwinit command not found. Please complete prerequisites first." throw "mwinit not available" } Install-GandalfMCP Write-Header "Gandalf Installation Complete!" Write-Success "Gandalf MCP has been installed and configured for Kiro" Write-Info "Restart Kiro to see the changes" } "3" { # Excel Theme Only Write-Host "Installing Excel Theme..." -ForegroundColor Cyan Write-Host "" Install-ExcelTheme Write-Header "Theme Installation Complete!" Write-Info "Restart Kiro if the theme doesn't appear immediately" } "4" { # Perform Daily Setup Write-Header "Daily Midway Setup" Write-Info "Running daily Midway authentication..." try { $mwinitCmd = Get-Command mwinit -ErrorAction SilentlyContinue if (-not $mwinitCmd) { Write-Error-Custom "mwinit command not found" Write-Info "Please ensure mwinit is installed on your system" throw "mwinit not available" } Write-Action "Running: mwinit -o -s" Write-Warning-Custom "You will be prompted to enter your password" Write-Warning-Custom "Note: When entering your password, you won't see it on screen - this is normal" Write-Host "" & mwinit -o -s if ($LASTEXITCODE -eq 0) { Write-Success "Midway authentication completed successfully!" Write-Info "Your Midway session is valid for 24 hours" } else { Write-Warning-Custom "mwinit exited with code: $LASTEXITCODE" Write-Info "If authentication succeeded, you can ignore this warning" } } catch { Write-Error-Custom "Midway authentication failed: $_" Write-Info "Please try running 'mwinit -o -s' manually" } # Show reminders and troubleshooting Write-Host "" Read-Host "Press Enter to view troubleshooting tips" Show-DailySetupReminder Show-TroubleshootingInfo Write-Host "" Read-Host "Press Enter to close this window" exit 0 } "5" { # Exit Write-Info "Exiting setup script" exit 0 } default { Write-Error-Custom "Invalid choice. Please run the script again and select 1-5." Read-Host "`nPress Enter to close this window" exit 1 } } # Pause and offer to join Slack community Write-Host "" Write-Info "Join the Slack group #gandalf-mcp-interest for updates and community support!" Read-Host "`nPress Enter to open Slack and close this window" # Open Slack channel in browser try { Start-Process "https://amazon.enterprise.slack.com/archives/C09P4PLRRJB" } catch { Write-Info "Could not open Slack automatically. Visit: https://amazon.enterprise.slack.com/archives/C09P4PLRRJB" } } catch { Write-Error-Custom "Setup failed: $_" Write-Info "Check the log file for details: $script:LogFile" Write-Info "Join Slack channel #gandalf-mcp-interest for assistance" # Pause to prevent window from closing on error Read-Host "`nPress Enter to close this window" exit 1 } } # Run the main function Main
